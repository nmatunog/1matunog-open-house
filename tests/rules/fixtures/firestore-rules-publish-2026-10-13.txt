rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ---- Open House (OH) ----
    function ohSignedIn() { return request.auth != null; }
    function ohEmail() { return request.auth.token.email.lower(); }
    function ohIsAdmin() {
      return ohSignedIn() && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/oh_admins/$(ohEmail()))
        && get(/databases/$(database)/documents/oh_admins/$(ohEmail())).data.active == true
        && get(/databases/$(database)/documents/oh_admins/$(ohEmail())).data.role == 'admin';
    }
    function ohIsViewer() {
      return ohIsAdmin() || (ohSignedIn() && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/oh_admins/$(ohEmail()))
        && get(/databases/$(database)/documents/oh_admins/$(ohEmail())).data.active == true);
    }
    function ohEvent(eid) { return get(/databases/$(database)/documents/oh_events/$(eid)).data; }

    match /oh_admins/{email} { allow read: if ohSignedIn() && ohEmail() == email; allow write: if false; } // console-only
    match /oh_config/public { allow read: if true; allow write: if ohIsAdmin(); }
    match /oh_people/{pid}     { allow read: if ohIsViewer(); allow write: if ohIsAdmin(); }
    match /oh_phoneIndex/{p}   { allow read: if ohIsViewer(); allow write: if ohIsAdmin(); }

    match /oh_events/{eid} {
      // guests need name, booths (code hashes), status and window only
      allow read: if true;
      allow write: if ohIsAdmin();

      match /checkins/{pid} {
        allow read: if ohIsViewer();
        allow create: if ohSignedIn()
          && ohEvent(eid).status == 'open'
          && request.time >= ohEvent(eid).checkInOpensAt && request.time <= ohEvent(eid).checkInClosesAt
          && request.resource.data.keys().hasOnly(['name','email','phone','invitedBy','stampsCompleted','checkedInAt','anonUid','consentVersion'])
          && request.resource.data.anonUid == request.auth.uid
          && request.resource.data.checkedInAt == request.time
          && request.resource.data.name is string && request.resource.data.name.size() > 1 && request.resource.data.name.size() <= 80
          && request.resource.data.email is string && request.resource.data.email.matches('^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$') && request.resource.data.email.size() <= 120
          && request.resource.data.phone is string && request.resource.data.phone.matches('^\\+639[0-9]{9}$')
          && request.resource.data.invitedBy is string && request.resource.data.invitedBy.size() <= 80
          && request.resource.data.stampsCompleted is int && request.resource.data.stampsCompleted >= 0 && request.resource.data.stampsCompleted <= 20
          && request.resource.data.consentVersion is string;
        allow update, delete: if ohIsAdmin();     // create-only for guests => one entry per person per event
      }
      match /draws/{n}    { allow read: if ohIsViewer(); allow create, update: if ohIsAdmin(); allow delete: if false; }
      match /winners/{pid} {
        allow read: if ohIsViewer();
        allow create: if ohIsAdmin() && !exists(/databases/$(database)/documents/oh_events/$(eid)/winners/$(pid));
        allow update: if ohIsAdmin(); // e.g. mark no_show releases lock: status field
        allow delete: if false;
      }

      // Plain tokens live here. Guests must never list or read this collection.
      match /stationSecrets/{boothId} {
        allow read, write: if ohIsAdmin();
      }

      // Doc id = sha256(eventId + token or short code). Get-by-id is enough to
      // prove the guest scanned or typed a real station secret. Listing would
      // leak every hash, so only admins may list.
      match /stampKeys/{keyId} {
        allow get: if true;
        allow list: if ohIsAdmin();
        allow create, update, delete: if ohIsAdmin();
      }

      // One stamp doc per device (anon uid) per station. Count these docs for
      // raffle eligibility — never trust checkins.stampsCompleted.
      match /stamps/{stampId} {
        allow get: if ohIsViewer()
          || (ohSignedIn() && stampId.matches('^' + request.auth.uid + '_.*'));
        allow list: if ohIsViewer();
        allow create: if ohSignedIn()
          && ohEvent(eid).status == 'open'
          && request.time >= ohEvent(eid).checkInOpensAt && request.time <= ohEvent(eid).checkInClosesAt
          && request.resource.data.keys().hasOnly(['boothId','anonUid','personId','stampedAt','stampKeyId'])
          && request.resource.data.anonUid == request.auth.uid
          && stampId == request.auth.uid + '_' + request.resource.data.boothId
          && request.resource.data.stampedAt == request.time
          && request.resource.data.boothId is string
          && request.resource.data.boothId.size() > 0
          && request.resource.data.boothId.size() <= 40
          && request.resource.data.personId is string
          && request.resource.data.personId.size() > 0
          && request.resource.data.personId.size() <= 80
          && request.resource.data.stampKeyId is string
          && request.resource.data.stampKeyId.size() == 64
          && exists(/databases/$(database)/documents/oh_events/$(eid)/checkins/$(request.resource.data.personId))
          && get(/databases/$(database)/documents/oh_events/$(eid)/checkins/$(request.resource.data.personId)).data.anonUid == request.auth.uid
          && exists(/databases/$(database)/documents/oh_events/$(eid)/stampKeys/$(request.resource.data.stampKeyId))
          && get(/databases/$(database)/documents/oh_events/$(eid)/stampKeys/$(request.resource.data.stampKeyId)).data.boothId == request.resource.data.boothId
          && get(/databases/$(database)/documents/oh_events/$(eid)/stampKeys/$(request.resource.data.stampKeyId)).data.eventId == eid;
        allow update, delete: if ohIsAdmin();
      }
    }
    // Legacy app: lock down. Admins can still read/export the old data for migration.
    match /artifacts/{appId}/public/data/{document=**} {
      allow read: if ohIsViewer();
      allow write: if ohIsAdmin();
    }
    // ---- end OH ----

    // Event Lead Gen (Netlify) — kept exactly as live on 2026-10-13
    match /artifacts/{appId}/public/data/leads/{leadId} {
      allow read, create, update, delete: if request.auth != null
        && appId == 'aia-lead-capture-2026';
    }

    // 1CMA Candidate Assessment (Interview Matrix) — kept exactly as live on 2026-10-13
    match /users/{userId}/profiles/{profileId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Block everything else
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
